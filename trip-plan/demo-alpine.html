💻 Code Example (Alpine.js)

// Register global component with Alpine.data() document.addEventListener('alpine:init', () => { Alpine.data('demoApp', () => ({ // 1. Data properties (like ref) name: 'Hiếu', firstName: 'Nguyễn', lastName: 'Hiếu', message: '', count: 0, isActive: false, hasError: false, // 2. Computed properties (with get) get fullName() { return `${this.firstName} ${this.lastName}` }, get messageLength() { return this.message.length }, get statusText() { if (this.isActive && this.hasError) return 'Active + Error' if (this.isActive) return 'Active' if (this.hasError) return 'Error' return 'Normal' }, // 3. Methods increment() { this.count++ }, decrement() { this.count-- }, reset() { this.count = 0 }, toggleActive() { this.isActive = !this.isActive }, toggleError() { this.hasError = !this.hasError }, resetStatus() { this.isActive = false this.hasError = false } })) })
