💻 Code Example (Element Plus + Vue 3 Composition API)

// Import Vue 3 và Element Plus từ CDN <script src="https://unpkg.com/vue@3.5.13/dist/vue.global.js"></script> <script src="https://unpkg.com/element-plus@2.13.0/dist/index.full.js"></script> // Setup app const { createApp, ref, computed } = Vue const app = createApp({ setup() { // Reactive state const fullName = ref('Nguyễn Văn A') const count = ref(50) // Computed const progressColor = computed(() => { if (count.value < 30) return '#f56c6c' if (count.value < 70) return '#e6a23c' return '#67c23a' }) // Methods const showMessage = (type, message) => { ElementPlus.ElMessage({ type, message }) } return { fullName, count, progressColor, showMessage } } }) // Use Element Plus app.use(ElementPlus) app.mount('#app')
