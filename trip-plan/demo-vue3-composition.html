💻 Code Example (Composition API)

// Import Vue 3 from CDN const { createApp, ref, computed } = Vue // Create app with Composition API createApp({ setup() { // 1. Reactive refs (dùng ref()) const name = ref('Hiếu') const firstName = ref('Nguyễn') const lastName = ref('Hiếu') const message = ref('') const count = ref(0) const isActive = ref(false) const hasError = ref(false) // 2. Computed properties (dùng computed()) const fullName = computed(() => { return `${firstName.value} ${lastName.value}` }) const messageLength = computed(() => { return message.value.length }) const statusText = computed(() => { if (isActive.value && hasError.value) return 'Active + Error' if (isActive.value) return 'Active' if (hasError.value) return 'Error' return 'Normal' }) // 3. Functions const increment = () => { count.value++ } const decrement = () => { count.value-- } const reset = () => { count.value = 0 } const toggleActive = () => { isActive.value = !isActive.value } const toggleError = () => { hasError.value = !hasError.value } const resetStatus = () => { isActive.value = false hasError.value = false } // 4. Return for template return { name, firstName, lastName, message, count, isActive, hasError, fullName, messageLength, statusText, increment, decrement, reset, toggleActive, toggleError, resetStatus } } }).mount('#app')
