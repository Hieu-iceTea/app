💻 Code Example (Composition API)

// Create app with setup function (Vue 3 style) const { ref, watchEffect, createApp } = MiniVue createApp(() => { // Declare refs const name = ref('Hiếu') const count = ref(0) // Computed with watchEffect const doubled = ref(0) watchEffect(() => { doubled.value = count.value * 2 }) // Functions function increment() { count.value++ } // Return for template binding return { name, count, doubled, increment } }).mount()
