💻 Code Example (Composition API)

// Create app with setup function (Vue 3 style) const { ref, watchEffect, createApp, classNames } = MiniVue createApp(() => { // 1. Declare refs const name = ref('Hiếu') const count = ref(0) // 2. Computed with watchEffect const doubled = ref(0) watchEffect(() => { doubled.value = count.value * 2 }) // 3. Functions function increment() { count.value++ } // 4. Dynamic classes (object syntax) const isActive = ref(false) const statusClass = ref({}) watchEffect(() => { statusClass.value = { active: isActive.value } }) // Or use classNames helper const manualClass = ref('') watchEffect(() => { manualClass.value = classNames({ active: isActive.value }) }) // Return for template binding return { name, count, doubled, increment, statusClass, manualClass } }).mount()
